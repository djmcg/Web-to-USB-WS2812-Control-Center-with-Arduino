📄 Kod firmware (sketch.ino) 
#include 

#define MAX_LEDS 256
#define DEFAULT_PIN 3

CRGB leds[MAX_LEDS];

uint16_t numLeds = 8;
uint8_t currentPin = DEFAULT_PIN;
uint8_t globalBrightness = 200;

uint8_t currentMode = 0;
uint8_t targetR = 255, targetG = 0, targetB = 0;
uint16_t animSpeed = 50;
uint16_t demoDuration = 10;

uint8_t gHue = 0;
unsigned long lastAnimUpdate = 0;
unsigned long lastDemoSwitch = 0;
uint8_t demoStep = 2;

void setup() {
  Serial.begin(9600);
  FastLED.addLeds(leds, MAX_LEDS);
  FastLED.setBrightness(globalBrightness);
  FastLED.clear(true);
}

void loop() {
  readSerialData();
  if (millis() - lastAnimUpdate >= animSpeed) {
    lastAnimUpdate = millis();
    gHue++;
    runEffectLogic();
    for (int i = numLeds; i < MAX_LEDS; i++) {
      leds[i] = CRGB::Black;
    }
    FastLED.show();
  }
}

void readSerialData() {
  while (Serial.available() >= 11) {
    if (Serial.read() == 0xAA) {
      currentMode = Serial.read();
      globalBrightness = Serial.read();
      targetR = Serial.read();
      targetG = Serial.read();
      targetB = Serial.read();
      uint8_t speedH = Serial.read();
      uint8_t speedL = Serial.read();
      animSpeed = (speedH << 8) | speedL;
      demoDuration = Serial.read();
      uint8_t newLeds = Serial.read();
      uint8_t newPin = Serial.read();
      if (newLeds > 0 && newLeds <= MAX_LEDS) {
        numLeds = newLeds;
      }
      if (newPin >= 2 && newPin <= 8 && newPin != currentPin) {
        currentPin = newPin;
        switch (currentPin) {
          case 2: FastLED.addLeds(leds, MAX_LEDS); break;
          case 3: FastLED.addLeds(leds, MAX_LEDS); break;
          case 4: FastLED.addLeds(leds, MAX_LEDS); break;
          case 5: FastLED.addLeds(leds, MAX_LEDS); break;
          case 6: FastLED.addLeds(leds, MAX_LEDS); break;
          case 7: FastLED.addLeds(leds, MAX_LEDS); break;
          case 8: FastLED.addLeds(leds, MAX_LEDS); break;
        }
        FastLED.setBrightness(globalBrightness);
        FastLED.clear(true);
      }
      FastLED.setBrightness(globalBrightness);
    }
  }
}

void runEffectLogic() {
  uint8_t activeMode = currentMode;
  if (currentMode == 1) {
    if (millis() - lastDemoSwitch >= ((unsigned long)demoDuration * 1000)) {
      lastDemoSwitch = millis();
      demoStep++;
      if (demoStep > 13) demoStep = 2;
    }
    activeMode = demoStep;
  }

  switch (activeMode) {
    case 0: fill_solid(leds, numLeds, CRGB(targetR, targetG, targetB)); break;
    case 2: fill_rainbow(leds, numLeds, gHue, 7); break;
    case 3: fill_rainbow(leds, numLeds, gHue, 7);
      if (random8() < 80) leds[random16(numLeds)] += CRGB::White;
      break;
    case 4: fadeToBlackBy(leds, numLeds, 10);
      leds[random16(numLeds)] += CHSV(gHue + random8(64), 200, 255);
      break;
    case 5: fadeToBlackBy(leds, numLeds, 20);
      leds[beatsin16(13, 0, numLeds - 1)] += CHSV(gHue, 255, 255);
      break;
    case 6: {
      uint8_t BeatsPerMinute = 62;
      CRGBPalette16 palette = PartyColors_p;
      uint8_t beat = beatsin8(BeatsPerMinute, 64, 255);
      for (int i = 0; i < numLeds; i++) {
        leds[i] = ColorFromPalette(palette, gHue + (i * 2), beat - gHue + (i * 10));
      }
      break;
    }
    case 7: fadeToBlackBy(leds, numLeds, 20);
      for (int i = 0; i < 4; i++) {
        leds[beatsin16(i + 7, 0, numLeds - 1)] |= CHSV(dothue, 200, 255);
        dothue += 32;
      }
      break;
    case 8: runFireEffect(); break;
    case 9: {
      uint8_t breath = beatsin8(15, 10, 255);
      fill_solid(leds, numLeds, CRGB(
        (targetR * breath) / 255,
        (targetG * breath) / 255,
        (targetB * breath) / 255
      ));
      break;
    }
    case 10: {
      static uint8_t step = 0;
      fadeToBlackBy(leds, numLeds, 100);
      for (int i = step; i < numLeds; i += 3) {
        leds[i] = CRGB(targetR, targetG, targetB);
      }
      step = (step + 1) % 3;
      break;
    }
    case 11: fadeToBlackBy(leds, numLeds, 15);
      if (random8() < 50) leds[random16(numLeds)] = CRGB(targetR, targetG, targetB);
      break;
    case 12: {
      CRGBPalette16 palette = OceanColors_p;
      for (int i = 0; i < numLeds; i++) {
        leds[i] = ColorFromPalette(palette, gHue + (i * 8), 255, LINEARBLEND);
      }
      break;
    }
    case 13: {
      static bool strobeOn = false;
      strobeOn = !strobeOn;
      if (strobeOn) fill_solid(leds, numLeds, CRGB(targetR, targetG, targetB));
      else fill_solid(leds, numLeds, CRGB::Black);
      break;
    }
  }
}

void runFireEffect() {
  static uint8_t heat[MAX_LEDS];
  for (int i = 0; i < numLeds; i++) {
    heat[i] = qsub8(heat[i], random8(0, ((55 * 10) / numLeds) + 2));
  }
  for (int k = numLeds - 1; k >= 2; k--) {
    heat[k] = (heat[k - 1] + heat[k - 2] + heat[k - 2]) / 3;
  }
  if (random8() < 120) {
    int y = random8(7);
    heat[y] = qadd8(heat[y], random8(160, 255));
  }
  for (int j = 0; j < numLeds; j++) {
    CRGB color = HeatColor(heat[j]);
    leds[j] = color;
  }
}
